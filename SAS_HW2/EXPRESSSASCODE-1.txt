/*Creat interaction term and quadratic terms*/
data express2;
  set EXPRESS;
  x1=Weight;
  x2=Distance;
  x1x2=Weight*Distance;
  x1SQ=Weight*Weight;
  x2SQ=Distance*Distance;
run;
proc print data=express2;
run;
proc sgplot data=express2;
  scatter y=Cost x=Weight;
run;

/*Regression of Reduced model*/
proc reg data=express2;
  model Cost=x1 x2 x1x2;
  title'Reduced model';
run;
/*Regression of complete model*/
proc reg data=express2;
  model Cost=x1 x2 x1x2 x1SQ  x2SQ;
  title'Full model';
run;

/*Partial F-test*/
proc reg data=express2;
  model Cost=x1 x2 x1x2 x1SQ  x2SQ;
  quadratic: test x1SQ, x2SQ=0; /*TEst if the quadratic terms significant*/
  title'Partial F-test';
run;
